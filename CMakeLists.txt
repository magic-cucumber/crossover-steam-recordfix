cmake_minimum_required(VERSION 3.20)
project(crossover_steam_recordfix VERSION 0.1.0 LANGUAGES C)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 target required")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GENERATED ${CMAKE_CURRENT_BINARY_DIR}/generated)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_thunks.py ${GENERATED} COMMAND_ERROR_IS_FATAL ANY)
if(MSVC)
  enable_language(ASM_MASM)
  set(THUNKS ${GENERATED}/thunks.asm)
else()
  enable_language(ASM)
  set(THUNKS ${GENERATED}/thunks.S)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
add_library(steam_recordfix SHARED src/recordfix.c ${THUNKS})
target_include_directories(steam_recordfix PRIVATE third_party/minhook/include)
target_link_libraries(steam_recordfix PRIVATE minhook)
set_target_properties(steam_recordfix PROPERTIES PREFIX "")
add_executable(recordfix_smoke tests/smoke.c)
set(BUILD_TESTING OFF CACHE BOOL "Disable vendored LibYAML test executables" FORCE)
add_subdirectory(third_party/libyaml EXCLUDE_FROM_ALL)
add_executable(recordfix_preload src/launcher.c src/config.c)
target_link_libraries(recordfix_preload PRIVATE yaml)
target_compile_features(recordfix_preload PRIVATE c_std_11)
add_executable(recordfix_launch_target tests/launch_target.c)
